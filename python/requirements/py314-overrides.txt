# (ci/ci.sh compile_pip_dependencies_py314). The shared source files pin
# versions that have no cp314 wheel; without these, the py3.14 constraints
# drift into pins that fall back to a source build and fail in the images
#

# wheel. A `python_version < '3.14'` marker drops the package on 3.14 entirely

# source pins move past them.
